 * Description :
 * Cette fonction réalise un tri rapide (QuickSort) d'un tableau
 * d'entiers. Le premier élément du tableau est choisi comme pivot.
 * Les éléments plus petits que le pivot sont placés à gauche,
 * puis les deux parties du tableau sont triées récursivement.
 */


void iqsort0 ( int *a , int n )
{
	int i , j ; // i parcourt le tableau et j indique la position
		    // du dernier élément inférieur au pivot.

	if ( n <=1) // Si le tableau a une taille de 0 ou 1 il est déjà trié
		return ; 

	// On choisit le premier élément a[0]
	for ( i = 1 , j = 0; i < n ; i ++)

		// Si l'élément a[i] est inférieur au pivot,
        	// on augmente j et on échange a[i] avec a[j].
        	// Ainsi, les éléments plus petits que le pivot
        	// sont progressivement regroupés à gauche.
		if ( a [ i ] < a [0])
			swap (++ j , i , a );

	// On place maintenant le pivot à sa position définitive.
	swap (0 , j , a );

	// On applique récursivement le même algorithme
    	// sur la partie gauche du pivot.
	iqsort0 (a , j );

	// On applique récursivement le même algorithme
    	// sur la partie droite du pivot.
	iqsort0 ( a + j +1 , n -j -1);
}
